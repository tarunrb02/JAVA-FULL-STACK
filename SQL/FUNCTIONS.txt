FUNCTIONS: ARE BUILT IN PROGRAMS WHICH ARE USED TO PERFORM SPECIFIC TASKS.
----------
	SELECT FUNCTION(COLUMN_NAME)  FROM TABLE_NAME;

2 TYPES OF FUNCTION 
	+ SINGLE ROW FUNCTION
	+ MULTI ROW FUNCTION

+ SINGLE ROW FUNCTION: IT WILL PERFORM THE FUNCTIONS ON EACH ROW AT A TIME AND RETURN THE RESULT FOR EACH ROW

5 TYPES OF SINGLE ROW FUNCTION
++ STRING FUNCTIONS:
	+++ UPPER(): IT WILL CONVERT THE LOWER CASE CHARACTER INTO UPPER CASE
		SELECT UPPER(FNAME) FROM EMP_LOG;
		SELECT *, UPPER(FNAME) FROM EMP_LOG;
		SELECT UPPER('tarunrb');

	+++ LOWER(): IT WILL CONVERT THE UPPER CASE CHARACTER INTO LOWER CASE
		SELECT LOWER(FNAME) FROM EMP_LOG;
		SELECT *, LOWER(FNAME) FROM EMP_LOG;
		SELECT LOWER('TARUNRB');

	+++ LENGTH(): IT RETURNS THE LENGTH OF STRING
		SELECT LENGTH(FNAME) FROM EMP_LOG;
		SELECT *, LENGTH(FNAME) FROM EMP_LOG WHERE E_ID=1;
		SELECT LENGTH('TARUNRB');

	+++ CONCAT(): IT RETURNS THE LENGTH OF STRING
		SELECT CONCAT(FNAME, LNAME) FROM EMP_LOG;
		SELECT *, LENGTH(FNAME, LNAME) FROM EMP_LOG WHERE E_ID=1;
		SELECT LENGTH('TARUNRB');
		SELECT CONCAT(FNAME,' ', LNAME, 'AGE IS ', E_AGE) FROM EMP_LOG LIMIT 3;
		+--------------------------------------------+
		| CONCAT(FNAME,' ', LNAME, 'AGE IS ', E_AGE) |
		+--------------------------------------------+
		| R@M KUMARAGE IS 23                         |
		| S$HANA PATILAGE IS 24                      |
		| PRIT#AM PARKERAGE IS 25                    |
 		+--------------------------------------------+

		Note: if any value in concatenation is null entire String will be NULL

	+++ SUBSTR():RETURNS THE PART OF(SUBSTRING) MAIN STRING
		SELECT SUBSTR(STRING, POSITION, NO_OF_CHAR);
		SELECT SUBSTR('DESTINATION', 3, 4);

	+++ INSTR(): IT RETURNS THE POSITION OF SUB STRING
		SELECT INSTR(STRING, SUB_STRING);
		SELECT('TARUNRB', 'RB');  // 6

	+++ REPLACE(): IT IS USED TO REPLACE THE NEW STRING WITH OLD STRING  
		SELECT REPLACE(STRING, OLD_STRING, NEW_STRING);
		SELECT REPLACE('TARUNRB','RB', ' R B');  //TARUN R B
	
	+++ TRIM: IT WILL REMOVE THE WHITE SPACE BEFORE AND AFTER THE STRING, NOT IN BETWEEN STRING
		SELECT TRIM(STRING);
		SELECT TRIM('    DESTINATION      CARRIER      ');   //DESTINATION CARRIER

	+++ REVERSE: REVERSE THE STRING.
		SELECT REVERSE(STRING);
		SELECT REVERSE('DESTINATION'); //  NOITANITSED

++ MATH FUNCTIONS: 
	+++ ROUND(): IT WILL ROUND OF THE VALUE UP TO THE SPECIFIED NUMBER OD DIGITES
		SELECT ROUND(DECIMAL ,NO_DECIMAL);
		SELECT ROUND(367.36895267, 3);    //367.369 
	
	+++ TRUNCATE():
		SELECT TRUNCATE(DECIMAL ,NO_DECIMAL);
		SELECT TRUNCATE(367.36895267, 3);    //367.368

	+++ CEIL(): IT WILL RETURNS THE SAME OR NEXT HIGHEST VALUE
		SELECT CEIL(VALUE);
		SELECT CEIL(568.3);    //569

	+++ FLOOR(): IT WILL RETURNS THE SAME OR NEXT LOWEST VALUE
		SELECT FLOOR(VALUE);
		SELECT FLOOR(568.3);    //568

	+++ ABS(): IT WILL GIVE POSITEVE VALUE

	+++ MOD(): IT WILL GIVE REMINDER VALUE.
		SELECT MOD(DIVIDEND, DIVISER);
		SELECT MOD(58735, 43);
	
	+++ POWER(): RETURNS EXPONETIAL VALUE
		SELECT POWER(BASE, POWER);
		SELECT POWER(2, 13);

	+++ SQRT(): RETURNS SQURE ROOT VALUE
		SELECT POWER(BASE, POWER);
		SELECT POWER(2, 13);

++ DATE FUNCTIONS():
	+++ CURDATE(): RETURN CURRENT ONLY DATE
		SELECT CURDATE();

	+++ CURTIME():RETURN CURRENT ONLY TIME
		SELECT CURTIME();

	+++ NOW(): RETURN CURRENT DATE AND TIME
		SELECT NOW();

	+++ DAYNAME(): RETURN CURRENT NAME OF THE DAY
		SELECT DAYNAME('2004-04-02');

	+++ DATEDIFF():
		SELECT DATEDIFF('2026-08-18', '2004-04-02');

	+++ TIMESTAMPDIFF():
		SELECT TIMESTAMPDIFF(YEAR, '2004-04-02', '2026-08-18');
		SELECT TIMESTAMPDIFF(MONTH, '2004-04-02', '2026-08-18');
		SELECT TIMESTAMPDIFF(DAY, '2004-04-02', '2026-08-18');

	+++ DATE_ADD():
		SELECT DATE_ADD('2020-10-07', INTERVAL 30 DAY);
		SELECT DATE_ADD('2020-10-07', INTERVAL 2 MONTH);
		SELECT DATE_ADD('2020-10-07', INTERVAL 2 YEAR);
	
	+++ DATE_SUB():
		SELECT DATE_SUB('2020-10-07', INTERVAL 4 MONTH);
		SELECT DATE_SUB('2020-10-07', INTERVAL 4 YEAR);

++ CONTROL FLOW FUNCTIONS:
	+++ IF(): RETURN ONE VALUE IF CONDITION IS TRUE OTHERWISE RETURN ANOTHER VALUE
		SELECT IF(CONDITION, VALUE1, VALUE2);
		SELECT IF(50=50, 'TRUE', 'FALSE');
		SELECT IF(50!=50, 'TRUE', 'FALSE');
		SELECT FNAME, E_AGE , IF(E_AGE<18, 'MINNOR', 'MAJOUR') AS AGE_STATUS FROM EMP_LOG;
		SELECT FNAME, SALARY , IF(SALARY>65000, 'HEIGH SALARY', 'LOW SALARY') AS SALARY_STATUS FROM EMP_LOG;

	+++ IFNULL(): IF THERE IS ANY NULL VALUE IT WILL REPLACE WITH NOT NULL VALUE
		SELECT IFNULL(NULL, VALUE);
		SELECT IFNULL(NULL, 673);
		SELECT IFNULL(EMP_EMAIL, 'EMAIL_IS NOT AVAILEBE') FROM EMP_LOG;

	+++ NULLIF(): RETURNS NULL IF BOTH THE VALUES ARE SAME OTHERWISE RETURN 1ST VALUE
		SELECT NULLIF(67,67); //NULL
		SELECT NULLIF(67,69); //67

	+++ CASE(): IT WILL CHECK MULTIPLE CONDITION AND CHOOSE ONE OPTION AMOUNG MANY
		CASE
		WHEN CODITION1 THEN VALUE1
		WHEN CODITION2 THEN VALUE2
		WHEN CODITION3 THEN VALUE3
		...
		...
		ELSE VALUE
		END;

+ AGGREGATE FUNCTION (MULTI ROW FUNCTION): IT WILL PERFORM THE FUNCTIONS ON ALL ROW AT A TIME AND RETURN ONLY ONE ROW AS THE RESULT 

	SELECT AGGREGATE_FUNCTIONS(COLUMN_NAME) FROM TABLE;

	DIFFERENT AGGREGATE FUNCTIONS:
	1. MAX(): returns the highest value from the column
	2. MIN(): returns the smallest value from the column
	3. AVG(): returns the average of all value from the column
	4. COUNT(): counts the number of values in the column
	5. SUM(): returns the sum of all value from the column

	SELECT MAX(SALARY) FROM EMP_LOG;
	+-------------+
	| MAX(SALARY) |
	+-------------+
	|   70000.000 |
	+-------------+

	SELECT COUNT(*) FROM EMP_LOG;

	SELECT COUNT(E_ID) FROM EMP_LOG;	

	SELECT COUNT(*), MAX(SALARY), MIN(SALARY), AVG(SALARY), SUM(SALARY) FROM EMP_LOG;
	+----------+-------------+-------------+---------------+-------------+
	| COUNT(*) | MAX(SALARY) | MIN(SALARY) | AVG(SALARY)   | SUM(SALARY) |
	+----------+-------------+-------------+---------------+-------------+
	|       13 |   70000.000 |   36000.000 | 47292.3076923 |  614800.000 |
	+----------+-------------+-------------+---------------+-------------+

HAVING CLAUSE: 
+ FILTERS GROUPED DATA AFTER AN AGGREGATION HAS OCCURRED. IT WORKS EXACTLY LIKE A WHERE CLAUSE, BUT INSTEAD OF FILTERING INDIVIDUAL ROWS, IT FILTERS ENTIRE GROUPS CREATED BY THE GROUP BY CLAUSE.

+ YOU CANNOT USE AGGREGATE FUNCTIONS (LIKE SUM(), COUNT(), AVG(), MAX(), OR MIN()) INSIDE A STANDARD WHERE CLAUSE. IF YOU TRY, THE DATABASE WILL THROW AN ERROR. THE HAVING CLAUSE WAS ADDED TO SQL SPECIFICALLY TO ALLOW FILTERING BASED ON THESE AGGREGATED CALCULATIONS.

+ SYNTAX: THE HAVING CLAUSE MUST ALWAYS BE PLACED AFTER THE GROUP BY CLAUSE AND BEFORE THE ORDER BY CLAUSE. 

	SELECT COLUMN_NAME, AGGREGATE_FUNCTION(COLUMN_NAME) FROM TABLE_NAME
	GROUP BY COLUMN_NAME
	HAVING AGGREGATE_FUNCTION(COLUMN_NAME) condition;
 

	Q1. DISPLAY THE DEPT WHERE TOTAL SALARY IS GREATER THAN 80000
