Q2. FIND THE AGES OF ALL THE EMPLOYEES AFTER 8 YEARS
	SELECT FNAME, lNAME, E_AGE, E_AGE+8 AS AFTER_8_YEARS FROM EMP_LOG;

Q3. FIND THE ANUAL SALARY OF EMPLOYEE
	SELECT FNAME, SALARY, SALARY*12 AS ANUAL_SALARY FROM EMP_LOG;

Q4. FIND THE DAY SALARY OF EMPLOYEES
	SELECT FNAME, SALARY, SALARY/30 AS DAY_SALARY FROM EMP_LOG;

Q5. INCREASE THE SALARY BY 15% TO ALL
	SELECT FNAME, SALARY, SALARY+(SALARY*15)/100 AS INCREMENTED_SALARY FROM EMP_LOG;

Q6. DECRIMENT THE SALARY BY 5% TO ALL 
	SELECT FNAME, SALARY, SALARY-(SALARY*5)/100 AS DECRIMENTED_SALARY FROM EMP_LOG;

Q7. REDUCE THE SALARY BY 8000 TO ID = 2;
	SELECT FNAME, SALARY, SALARY-8000 FROM EMP_LOG WHERE E_ID=2;

+ LIKE: USED FOR PATTEREN MATCHING

WE USE WILDCARD CHARECTER '%' AND '_'

'%' ---> MULTIPLE CHARECTER
'_' --> ONLY ONE CHARECTER

IF THE PATTERN IT SELF CONTAINS WILD CARD CHARECTER WE USE ESCAPE CHARECTER '\' , '#' , '$' ..... (USED TO REPRESENT THE NEXT CHAR IS NOT WILD CARD CHAR IT'S A NORMAL CHAR
	++ SELECT * FROM EMP_LOG WHERE FNAME LIKE "%\%%";

	SELECT * FROM EMP_LOG WHERE FNAME LIKE "%/%%" ESCAPE '/'
	SELECT * FROM EMP_LOG WHERE FNAME LIKE "%/_%" ESCAPE '/';

ORDER BY CLAUSE (CLAUSE => CONDITION)

	SYN: ORDER BY COLUMN_NAME DESC/ASC; 
	IT IS CLASUE WHICH USED TO SORT/ ARRANGE THE DATA IN ADCENDING OR DESCENDING ORDER.

+ ASC - KEY WORD USED TO SORT THE DATA IN ASCENDING ORDER
+ DESC - KEY WORD USED TO SORT THE DATA IN DESCENDING ORDER

+ BY DEFAULT DATA WILL BE SORTED IN ASCENDING ORDER

LIMIT -- HOW MANY ROWS ARE REQUIRED (STARTS FROM 1)

OFFSET -- HOW MANY ROWS NEED TO BE SKIP (STARTS FROM 0) 


Q1. ARRAANGE AND DISPLAY THE SALARIES IN DESCENDING ORDER
	SELECT * FROM EMP_LOG ORDER BY SALARY DESC;

Q2. DISPLAY THE LOWEST SALARY AT TOP.
