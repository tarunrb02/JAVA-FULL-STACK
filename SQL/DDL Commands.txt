DDL: It is used to create or alter the table, and it will work only on the table Structure not with the data inside the table
	
	+ CREATE: CREATE TABLE TABLE_NAME(COL1 DT, COL2 DT);
	+ ALTER: use to add/drop a column, add/drop constrains, rename column(RENAME COLUMN), change data type (MODIFY)	

+ TO ADD A COLUMN INTO THE EXISTING TABLE
	  ALTER TABLE TABLE_NAME ADD COLUMN_NAME DATA_TYPE(SIZE)
	  EX:  ALTER TABLE EMPLOYEE ADD SALARY DECIMAL(10,3);

+ TO ADD A COLUMN INTO THE EXISTING TABLE AFTER COLUMN_NAME
	  ALTER TABLE EMPLOYEE ADD PHONE_NO BIGINT AFTER E_AGE;
	
+ TO ADD A COLUMN INTO THE EXISTING TABLE AT FIRST POSITIION
	  ALTER TABLE EMPLOYEE ADD GENDER CHAR(1) FIRST;

+ CHANGE THE NAME (RENAME COLUMN) OF EXISTING COLUMN NAME FROM THE TABLE
	  ALTER TABLE TABLE_NAME RENAME COLUMN OLD_COL_NAME TO NEW_COL_NAME DATA_TYPE;
	  EX: ALTER TABLE EMPLOYEE RENAME COLUMN EMAIL TO EMP_EMAIL;

+ CHANGE THE DATATYPE OF EXISTING COLUMN FROM THE TABLE
	  ALTER TABLE TABLE_NAME MODIFY COLUMN_NAME NEW_DATATYPE


COMMANDS TO ADD CONSTRAINS TO COLUMNS OF EXISTIG TABLE
------------------------------------------------------

1. ALTER TABLE STDLOG ADD UNIQUE(EMAIL);                              // ADD UNIQE
   ALTER TABLE TABLE_NAME ADD UNIQUE(COLUMN_NAME);

2. ALTER TABLE STDLOG ADD PRIMARY KEY(STD_ID);                        // ADD PRIMARY KEY
   ALTER TABLE TABLE_NAME ADD PRIMARY KEY(COLUMN_NAME);

3. ALTER TABLE STDLOG MODIFY FNAME VARCHAR(30) NOT NULL;               // ADD NOT NULL similar to adding data type 
   ALTER TABLE TABLE_NAME  MODIFY COLUMN_NAME DATATYPE NOT NULL;

4. ALTER TABLE STDLOG ADD CHECK(S_AGE<=19);                           // ADD CHECK
   ALTER TABLE TABLE_NAME ADD CHECK(condition on COLUMN_NAME);

5. ALTER TABLE STDLOG ADD FOREIGN KEY(C_ID) REFERENCES COURSE(C_ID);  // ADD FOREIGN KEY
   ALTER TABLE TABLE_NAME ADD FOREIGN KEY(COLUMN_NAME) REFERENCES PARENT_TABLE(COLUMN_NAME);

6. ALTER TABLE STDLOG ALTER COLUMN CLASS SET DEFAULT 12;              // SET DEFAULT
   ALTER TABLE TABLE_NAME ALTER COLUMN COLUMN_NAME SET DEFAULT VALUE;
	

COMMANDS TO DROP CONSTRAINS TO COLUMNS OF EXISTIG TABLE
-------------------------------------------------------

1. ALTER TABLE STDLOG DROP INDEX EMAIL;                 	// DROP UNIQE
   ALTER TABLE TABLE_NAME ADD INDEX COLUMN_NAME;

2. ALTER TABLE STDLOG DROP PRIMARY KEY;                 	// DROP PRIMARY
   ALTER TABLE TABLE_NAME ADD PRIMARY KEY(COLUMN_NAME);

3. ALTER TABLE STDLOG MODIFY NAME VARCHAR(30) NULL;     	// REMOVE NOT NULL
   ALTER TABLE TABLE_NAME  MODIFY COLUMN_NAME DATATYPE NOT NULL;

4. ALTER TABLE STDLOG DROP CHECK stdlog_chk_1;          	//DROP CHECK
   ALTER TABLE STDLOG DROP CHECK Constrain_Name;

5. ALTER TABLE STDLOG DROP FOREIGN KEY stdlog_ibfk_1;   	// DROP FOREIGN KEY
   ALTER TABLE STDLOG DROP FOREIGN KEY Constrain_Name;

6. ALTER TABLE STDLOG ALTER COLUMN CLASS DROP DEFAULT;  	// DROP DEFAULT
   ALTER TABLE TABLE_NAME ALTER COLUMN COLUMN_NAME DROP DEFAULT;


+ RENAME COLUMN NAME
	ALTER TABLE EMPLOYEE RENAME COLUMN PHONE_NO TO EMP_PHONE_NUM;

+ RENAME TABLE NAME
	SYN: ALTER TABLE TABLE_NAME RENAME TO NEW_TABLE_NAME;
	mysql> ALTER TABLE EMPLOYEE RENAME TO EMP_LOG;

+ TO REMOVE ALL RECOREDS FROM TABLE
	TRUNCATE TABLE TABLE_NAME;

	mysql> TRUNCATE TABLE COURSE;
	Query OK, 0 rows affected (0.04 sec)

	mysql> SELECT * FROM COURSE;
	Empty set (0.00 sec)

+ TO REMOVE TABLE/ DATABASE
	DROP TABLE TABLE_NAME;
	DROP DATABASE DATABASE_NAME;

+ TO REMOVE SPECIFIC COLUMN FROM TABLE
	ALTER TABLE TABLE_NAME DROP COLUMN_NAME;
	EX: ALTER TABLE STDLOG DROP S_AGE;
