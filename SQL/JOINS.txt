JOINS: JOINS ARE USED TO COMBINE 2 OR MORE TABLES BY USING COMMON OR RELATED COLUMNS FROM BOTH THE TABLES.

	SYNTAX: SELECT COLUMNS FROM TAB1 (T1) TYPE OF JOIN TAB2 (T2) ON T1.COMMON_COL= T2.COMMON_COL

	TYPES OF JOINS:
	1. INNER JOIN: JOIN WHICH RETURNS ONLY MATCHING RECORDS FROM THE BOTH TABLE

	2. LEFT JOIN: IT RETURNS ALL THE RECORDS FROM LEFT TABLE AND ONLY MATCHING RECORDS FROM RIGHT TABLE
	
	3. RIGHT JOIN: IT RETURNS ALL THE RECORDS FROM RIGHT TABLE AND ONLY MATCHING RECORDS FROM LEFT TABLE

	4. CROSS JOIN: IT RETURNS ALL POSSIBLE COMBINATIONS FROM TABLES

	5. SELF JOIN: ANY ONE OF THE ROW FROM THE TABLE IS JOINNG WITH THE SAME TABLE

INNER JOIN:
	Q1. DISPLAY FNAME, EMP ID AND DEPT NAME OF EMP WHO ARE WORKING IN IT DEPT
	    SELECT E.FNAME, E.E_ID, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME='IT';

	Q2. DISPLAY ALL THE DETAILS OF EMP ALONG WITH THERE DEPT NAME
	    SELECT E.*, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID= D.D_ID;

	Q3. DISPLAY EMP ID, NAME AND SALARY ALONG WITH THEIR DEPT NAME AND ID
	    SELECT E.E_ID, E.FNAME, E.SALARY, D.D_ID, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID;

	Q4. DISPLAY EMP NAME AND DEPT NAME WHO ARE LOCATED IN BENGALURU
	    SELECT E.FNAME, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.EMP_LOCATION='BENGALURU';

	Q6. DISPLAY EMPS WHOSE SALARY IS GRATER THEN OF ARUN ALONG WITH THERE DEPT NAME
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.SALARY>(SELECT SALARY FROM EMP_LOG WHERE FNAME='ARUN');

	Q7. DISPLAY EMPS DETAILS IN DESC ORDER OF THEIR SALARY ALONG WITH THERE DEPT NAME
	SELECT E.*, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID ORDER BY E.SALARY DESC;

  	Q8. DISPLAY EMPS WHOSE SALARY IS GRATER THEN 60000 ALONG WITH THERE DEPT
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.SALARY>60000;

	Q9. DISPLAY EMPS NAME SALARY AND DEPT NAME WORKING IN IT OR HR
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME='IT' OR D.D_NAME='HR';

	Q10. DISPLAY THE EMPS WHOSE SALARY BETWEEN 50000 AND 80000 ALONG WITH THERE DEPT NAME
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.SALARY BETWEEN 50000 AND 80000;

	Q11. DISPALY HEIGEST PAID EMPS ALONG WITH THERE DEPT NAME
	SELECT E.*, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID ORDER BY E.SALARY DESC LIMIT 1;

	Q12. DISPLAY EMPS WHOSE DEPT START WITH I 
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME LIKE 'I%';

	Q13. DISPLAY EMPS WHO BELONG TO SAME DEPT AS RAVI
	SELECT E.FNAME, E.SALARY, D.D_NAME FROM EMP_LOG E INNER JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.DEPT_ID=(SELECT DEPT_ID FROM EMP_LOG WHERE FNAME='AFFAN');

	Q14. DISPLAY NAME DEPT NAME AND COUNT OF EMP IN THERE DEPT
	SELECT E.FNAME, D.D_NAME,(SELECT COUNT(E_ID) FROM EMP_LOG WHERE DEPT_ID=E.DEPT_ID) AS COUNT_EMP FROM EMP_LOG E INNER JOIN DEPT D WHERE E.DEPT_ID=D.D_ID;


LEFT JOIN AND RIGHT JOIN:

	Q1. DISPLAY ALL THE EMP WITH THERE DEPT NAME
	    SELECT E.*, D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID;

	Q2. DISPLAY ALL THE DEPT WITH THERE EMPLOYEE
	    SELECT E.*, D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID;
	
	Q3. DISPLAY ALL THE EMPS WHO ARE NOT ASSIGNED TO ANY DEPT.
            SELECT E.FNAME, D.D_NAME  FROM EMPS E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME IS NULL;

	Q4. DISPALY ALL THE DEPTS WITH THE NUMBER OF EMPS IN EACH DEPT INCLUDING DEPTS WITH ZERO EMPS .
    	    SELECT D.D_NAME, COUNT(E_ID) FROM EMPS E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID GROUP BY D.D_NAME;

	Q5. FIND THE DEPT EHAT HAVE NO EMP.
            SELECT D.D_NAME, COUNT(E_ID) FROM EMPS E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID GROUP BY D.D_NAME HAVING COUNT(E_ID)=0;
	Q6. DISPLAY ALL THE DEPT AND REPLACE NULL EMPLOYEE WITH 'NO EMP'.
            SELECT D.D_NAME, IFNULL(E.FNAME,'NO EMPLOYEE') FROM EMPS E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID;

	Q7. DISPLAY EACH EMPS ALONG WITH THEIR DEPT NAME AND SHOW EMPLOYEE WITH SALARY IS GRATER THEN THE AVERAGE SALARY 
            SELECT E.FNAME, D.D_NAME, E.SALARY FROM EMPS E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE SALARY>(SELECT AVG(SALARY) FROM EMPS);

	Q8. FIND AND DISPLAY THE DEPT WITH TOP 1(HIGHEST) SUM SALARY OF EMPS.
            SELECT D.*, SUM(E.SALARY) FROM EMPS E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID GROUP BY D.D_ID ORDER BY SUM(E.SALARY) DESC LIMIT 1;

	Q9. DISPLAY ALL THE EMPS & THEIR DEPT NAMES, BUT SHOW ONLY EMPS ERNING MORE THEN 60000 
            SELECT E.FNAME, D.D_NAME, E.SALARY FROM EMPS E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.SALARY>60000;

	Q10. DISPLAY ALL THE DEPT WHOSE TOTAL SALARY S GREATER THEN 10000
	     SELECT D.*, SUM(E.SALARY) FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID GROUP BY D.D_ID HAVING SUM(E.SALARY)>100000

	Q11 DISPLAY DEPT HAVING ATLISTE 2 EMPS
	SELECT D.D_NAME, COUNT(E_ID) FROM EMP_LOG E RIGHT JOIN DEPT D ON  E.DEPT_ID=D.D_ID GROUP BY D.D_NAME HAVING COUNT(E_ID)>=2;

FULL OUTER JOIN: LEFT JOIN UNION RIGHT JOIN

	Q1. DISPLAY ALL EMPs AND DEPTs ALONG WITH UNMATCHED RECORDS
		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID
    		UNION
    		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID;

	Q2. DISPLAY ONLY THE EMP WHO ARE NOT ASSIGNED TO ANY DEPT AND DEPT HAVING NO EMP
		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME IS NULL
    		UNION
    		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.FNAME IS NULL;

	Q3. DISPLAY ALL EMPs AND DEPTs, REPLACE NULL WITH MEANINGFULL TEXT 
		SELECT E.FNAME, IFNULL(D.D_NAME, 'NO DEPT') FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID
    		UNION
    		SELECT IFNULL(E.FNAME,'NO EMP'), D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID;

	Q4. DISPLAY ALL EMPs AND DEPTs AND SORT THEM BY DEPT_NAME;
		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID
    		UNION
		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID ORDER BY D_NAME

	Q5. DISPLAY ONLY UNMATCHED RECORDES FROM BOTH TABLES
		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE D.D_NAME IS NULL
    		UNION
    		SELECT E.FNAME,D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID WHERE E.FNAME IS NULL;

	Q6. DISPLAY EMPS NAME AGE AND THER DEPT NAME INCLUDING UNMATCHED RECORDES THAN  SORT THE RESULT BY AGE
		SELECT E.FNAME, E.E_AGE, D.D_NAME FROM EMP_LOG E LEFT JOIN DEPT D ON E.DEPT_ID=D.D_ID
   		UNION
   	 	SELECT E.FNAME, E.E_AGE, D.D_NAME FROM EMP_LOG E RIGHT JOIN DEPT D ON E.DEPT_ID=D.D_ID ORDER BY E_AGE;

CROSS JOIN: TURNS ALL POSIBLE RESULTS
	 SELECT E.FNAME, D.D_NAME FROM EMP_LOG E CROSS JOIN DEPT D;
	 SELECT S.STD_ID,C.* FROM STDLOG S CROSS JOIN COURSE C;

SELF JOIN: ANY ONE OF THE ROW FROM THE TABLE IS JOINNG WITH THE SAME TABLE

	Q1. FIND AND DISPLAY EMPS WHO ARE REFERED BY ANYONE
	SELECT E.FNAME AS EMP_NAME, R.FNAME AS REFF FROM EMP_LOG E LEFT JOIN EMP_LOG R ON E.REFFERD_BY=R.E_ID;

	Q2. FIND AND DISPLAY EMPS WHO ARE NOT REFERED BY ANYONE 
	SELECT E.FNAME AS EMP_NAME, R.FNAME AS REFF FROM EMP_LOG E LEFT JOIN EMP_LOG R ON E.REFFERD_BY=R.E_ID WHRE E.REFFERD BY IS NULL;

	Q3. FIND AND DISPLAY ONLY THE EMPS WHO REFERED BY OTHER EMPLOYEE
	SELECT E.FNAME, R.FNAME FROM EMP_LOG E LEFT JOIN EMP_LOG R ON E.REFFERD_BY=R.E_ID WHERE E.REFFERD_BY IS NOT NULL;
	SELECT E.FNAME, R.FNAME FROM EMP_LOG E INNER JOIN EMP_LOG R ON E.REFFERD_BY=R.E_ID;

	Q4. DISPLAY EMPS NAMES AND THERE MANAGER NAMES
       	SELECT E.FNAME AS EMPS, M.FNAME AS MANAGER FROM EMPS E LEFT JOIN EMPS M ON E.MANAGER=M.E_ID; 

	Q5 DISPLAY THE EMPS DOSE NOT HAVE ANY MANAGER 
      	SELECT E.FNAME AS EMPS, M.FNAME AS MANAGER FROM EMPS E LEFT JOIN EMPS M ON E.MANAGER=M.E_ID WHERE E.MANAGER IS NULL;
