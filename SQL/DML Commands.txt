DML: Data manipulation Language: it is used to update or modify the data inside the table

INSERT: It is used to insert/add the values into the table

	SYN1: INSERT INTO TABLE_NAME VALUES(.........);
	 INSERT INTO EMP_LOG VALUES (11, 'KUNAL', 'SHARMA', 22, 22222222222, 'KUNAL@GMAIL.COM', 'RAJASTHAN', 103, 50000);

	SYN2: INSERT INTO TABLE_NAME VALUES(.........),(.........),(.........)....;
	 INSERT INTO EMP_LOG VALUES (12, 'AMAN', 'GUPTA', 22, 21212121121, 'AMAN@GMAIL.COM', 'RAMPUR', 104, 50000),
    -> (13, 'KUSHAL', 'SHARMA', 22, 11111111111, 'KUSHAL@GMAIL.COM', 'RAJASTHAN', 104, 40000);

	SYN3: INSERT INTO TABLE_NAME(C0L1, COL2, COL3........) VALUES(.........);
	INSERT INTO EMP_LOG (E_ID, FNAME, LNAME, E_AGE, EMP_PHONE_NUM, EMP_EMAIL, EMP_LOCATION, DEPT_ID, SALARY) VALUES (14, 'NAVEEN', 'SHA', 22, 33333333333, 'NAVEEN@GMAIL.COM', 'BENGALRU', 101, 50000);

UPDATE: It is used to modify the records/row data

	SYN: UDATE TABLE_NAME
	     SET COUMN_NAME=VALUE
	     WHERE CONDITION;
        
	mysql> UPDATE EMP_LOG
	    -> SET EMP_EMAIL= 'RAMKUMAR@GMAIL.COM'
	    -> WHERE E_ID=1;

	+ MULTIPLE VALUES IN A SAME ROW
	mysql> UPDATE EMP_LOG
	    -> SET LNAME=NULL, EMP_EMAIL= 'AFFAN123@GMAIL.COM' WHERE E_ID=6;

	+ SYN: UPDATE MULTIPLE VALUES IN A SAME COLUMN
		UPDATE TABLE_NAME
		SET COLUMN_NAME= CASE
		WHEN CODITION1 THEN VALUE1
		WHEN CODITION2 THEN VALUE2
		WHEN CODITION3 THEN VALUE3
		...
		...
		ELSE VALUE
		END;

	mysql> UPDATE EMP_LOG SET SALARY = CASE
	    -> WHEN E_ID=1 THEN 35000
	    -> WHEN E_ID=2 THEN 50000
	    -> WHEN E_ID=3 THEN 70000
	    -> ELSE SALARY
	    -> END;

	mysql> UPDATE EMP_LOG SET SALARY = CASE
	    -> WHEN DEPT_ID=101 THEN 45000
	    -> WHEN DEPT_ID=102 THEN 70000
	    -> WHEN DEPT_ID=103 THEN 50000	
	    -> WHEN DEPT_ID=104 THEN 30000
	    -> WHEN DEPT_ID=105 THEN 40000
	    -> ELSE SALARY
	    -> END;

DELETE: IT WILL REMOVE A SPECIFIC ROW FROM THE TABLE
	SYN: DELETE FROM TABLE_NAME WHERE CONDITION;
	
	mysql> DELETE FROM EMP_LOG WHERE E_ID=14;  --------------------- REMOVE SINGLE ROW
	Query OK, 1 row affected (0.07 sec)

	mysql> DELETE FROM EMP_LOG WHERE E_ID IN (11,12,13); --------------- REMOVE MULTIPLE ROW
	Query OK, 3 rows affected (0.07 sec)
