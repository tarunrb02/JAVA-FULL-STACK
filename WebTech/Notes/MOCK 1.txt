---------+
| EMP_LOG | CREATE TABLE `emp_log` (
  `E_ID` int NOT NULL,
  `FNAME` varchar(30) NOT NULL,
  `LNAME` varchar(30) DEFAULT NULL,
  `E_AGE` int DEFAULT NULL,
  `EMP_PHONE_NUM` bigint DEFAULT NULL,
